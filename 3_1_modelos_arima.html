# Function to upload and read Excel files
def upload_and_read_excel():
    uploaded = files.upload()
    data = {}
    for filename in uploaded.keys():
        df = pd.read_excel(filename)
        if pd.api.types.is_datetime64_any_dtype(df['Local Date']):
            df['Date'] = df['Local Date']
        else:
            df['Date'] = pd.to_datetime(df['Local Date'], origin='1899-12-30', unit='D')
        df.set_index('Date', inplace=True)
        data[filename.split('.')[0]] = df['Close']
    return data

# Random Walk Test
def random_walk_test(series, name):
    plt.figure(figsize=(12, 6))
    plt.plot(series, label=f'{name} Price', color='blue')
    plt.title(f'{name} - Random Walk Plot')
    plt.xlabel('Date')
    plt.ylabel('Price')
    plt.legend()
    plt.grid(True)
    plt.savefig(f'{name}_random_walk.png')
    plt.show()
    plt.close()

    diff = series.diff().dropna()
    plt.figure(figsize=(12, 6))
    plt.plot(diff, label=f'{name} First Difference', color='green')
    plt.title(f'{name} - First Difference (Random Walk Check)')
    plt.xlabel('Date')
    plt.ylabel('Difference')
    plt.legend()
    plt.grid(True)
    plt.savefig(f'{name}_random_walk_diff.png')
    plt.show()
    plt.close()

    # Interpretation
    diff_std = diff.std()
    diff_mean = diff.mean()
    print(f'\nInterpretation of Random Walk Test for {name}:')
    print(f'- Price Plot: Shows the stock price trend over time. Visible trends or volatility suggest non-random behavior.')
    print(f'- First Difference Plot: The differences (std: {diff_std:.4f}, mean: {diff_mean:.4f}) should resemble white noise for a random walk.')
    if abs(diff_mean) < 0.01 and diff_std < series.std() / 10:
        print(f'- Result: The small mean and std of differences suggest {name} may follow a random walk, making price movements hard to predict.')
    else:
        print(f'- Result: The differences show patterns or high volatility, suggesting {name} may not strictly follow a random walk.')

# Unit Root Test (Augmented Dickey-Fuller)
def unit_root_test(series, name):
    result = adfuller(series)
    print(f'\nUnit Root Test (ADF) for {name}:')
    print(f'ADF Statistic: {result[0]:.4f}')
    print(f'p-value: {result[1]:.4f}')
    print('Critical Values:')
    for key, value in result[4].items():
        print(f'\t{key}: {value:.4f}')

    rolling_mean = series.rolling(window=20).mean()
    rolling_std = series.rolling(window=20).std()

    plt.figure(figsize=(12, 6))
    plt.plot(series, label='Original', color='blue')
    plt.plot(rolling_mean, label='Rolling Mean', color='red')
    plt.plot(rolling_std, label='Rolling Std', color='green')
    plt.title(f'{name} - Rolling Statistics')
    plt.xlabel('Date')
    plt.ylabel('Price')
    plt.legend()
    plt.grid(True)
    plt.savefig(f'{name}_unit_root.png')
    plt.show()
    plt.close()

    # Interpretation
    print(f'\nInterpretation of Unit Root Test for {name}:')
    print(f'- ADF Test: A low p-value (< 0.05) indicates stationarity (no unit root).')
    if result[1] < 0.05:
        print(f'- Result: p-value ({result[1]:.4f}) < 0.05, suggesting {name} is stationary. Suitable for direct modeling.')
    else:
        print(f'- Result: p-value ({result[1]:.4f}) >= 0.05, suggesting {name} is non-stationary. Differencing is needed (as done in ARIMA/SARIMA).')
    print(f'- Rolling Stats Plot: Stable mean and std indicate stationarity. Varying stats suggest trends or volatility.')

# Cointegration Test
def cointegration_test(series1, series2, name1, name2):
    df = pd.concat([series1, series2], axis=1).dropna()
    score, p_value, _ = coint(df.iloc[:, 0], df.iloc[:, 1])
    print(f'\nCointegration Test between {name1} and {name2}:')
    print(f'Cointegration Score: {score:.4f}')
    print(f'p-value: {p_value:.4f}')

    spread = df.iloc[:, 0] - df.iloc[:, 1]
    plt.figure(figsize=(12, 6))
    plt.plot(spread, label='Spread', color='purple')
    plt.title(f'Spread between {name1} and {name2}')
    plt.xlabel('Date')
    plt.ylabel('Spread')
    plt.legend()
    plt.grid(True)
    plt.savefig(f'{name1}_{name2}_cointegration.png')
    plt.show()
    plt.close()

    # Interpretation
    print(f'\nInterpretation of Cointegration Test between {name1} and {name2}:')
    print(f'- Cointegration Test: A low p-value (< 0.05) indicates a stable long-term relationship.')
    if p_value < 0.05:
        print(f'- Result: p-value ({p_value:.4f}) < 0.05, suggesting {name1} and {name2} are cointegrated. Suitable for pairs trading.')
    else:
        print(f'- Result: p-value ({p_value:.4f}) >= 0.05, suggesting no cointegration. Pairs trading may not be effective.')
    print(f'- Spread Plot: A mean-reverting spread (stable around a level) supports cointegration. Wide fluctuations suggest no stable relationship.')

# ARIMA Forecast (Price Changes)
def arima_forecast(series, name, order=(1,0,1)):
    try:
        diff_series = series.diff().dropna()

        plt.figure(figsize=(12, 4))
        plt.subplot(121)
        plot_acf(diff_series, ax=plt.gca(), lags=20)
        plt.title(f'{name} - ACF of Price Changes')
        plt.subplot(122)
        plot_pacf(diff_series, ax=plt.gca(), lags=20)
        plt.title(f'{name} - PACF of Price Changes')
        plt.tight_layout()
        plt.savefig(f'{name}_arima_acf_pacf.png')
        plt.show()
        plt.close()

        # Interpretation of ACF/PACF
        print(f'\nInterpretation of ACF/PACF for {name} (ARIMA):')
        print(f'- ACF/PACF Plots: Significant lags suggest the order of AR (p) and MA (q) terms.')
        print(f'- Current order (1,0,1): Assumes one AR and one MA term. Adjust if many lags are significant.')

        model = ARIMA(diff_series, order=order)
        results = model.fit()

        print(f'\nARIMA Model Summary for {name} (Price Changes):')
        print(results.summary())

        trading_minutes = 8 * 60
        forecast_diff = results.forecast(steps=trading_minutes)
        last_price = series.iloc[-1]
        forecast_prices = np.cumsum(forecast_diff) + last_price
        forecast_index = pd.date_range(start=series.index[-1] + timedelta(minutes=1),
                                     periods=trading_minutes, freq='T')

        plt.figure(figsize=(12, 6))
        plt.plot(series[-200:], label='Actual Prices', color='blue')
        plt.plot(forecast_index, forecast_prices, label='ARIMA Forecast', color='red')
        plt.title(f'{name} - ARIMA Price Forecast for April 30, 2025')
        plt.xlabel('Date')
        plt.ylabel('Price')
        plt.legend()
        plt.grid(True)
        plt.xticks(rotation=45)
        plt.tight_layout()
        plt.savefig(f'{name}_arima_forecast.png')
        plt.show()
        plt.close()

        plt.figure(figsize=(12, 6))
        plt.plot(forecast_index, forecast_diff, label='ARIMA Forecasted Price Changes', color='orange')
        plt.title(f'{name} - ARIMA Forecasted Price Changes for April 30, 2025')
        plt.xlabel('Date')
        plt.ylabel('Price Change')
        plt.legend()
        plt.grid(True)
        plt.xticks(rotation=45)
        plt.tight_layout()
        plt.savefig(f'{name}_arima_diff.png')
        plt.show()
        plt.close()

        # Interpretation of ARIMA Forecast
        print(f'\nInterpretation of ARIMA Forecast for {name}:')
        print(f'- Price Forecast: Predicts prices for April 30, 2025, based on minute-by-minute price changes.')
        print(f'- Forecast Trend: {"Upward" if forecast_prices.iloc[-1] > forecast_prices.iloc[0] else "Downward or Flat"} trend expected.')
        print(f'- Price Changes Plot: Shows volatility of predicted changes (std: {forecast_diff.std():.4f}). High volatility suggests uncertainty.')
        print(f'- Reliability: Good for short-term dynamics but may miss intraday seasonality. Check ACF/PACF for model fit.')

    except Exception as e:
        print(f'ARIMA forecast failed for {name}: {str(e)}')

# SARIMA Forecast (Price Changes)

# Prophet Forecast
def prophet_forecast(series, name):
    try:
        df_prophet = pd.DataFrame({'ds': series.index, 'y': series.values})
        model = Prophet(daily_seasonality=True, yearly_seasonality=False, weekly_seasonality=False)
        model.fit(df_prophet)

        trading_minutes = 8 * 60
        future = pd.date_range(start=series.index[-1] + timedelta(minutes=1),
                              periods=trading_minutes, freq='T')
        future_df = pd.DataFrame({'ds': future})

        forecast = model.predict(future_df)
        forecast_prices = forecast['yhat']

        plt.figure(figsize=(12, 6))
        plt.plot(series[-200:], label='Actual Prices', color='blue')
        plt.plot(future, forecast_prices, label='Prophet Forecast', color='purple')
        plt.title(f'{name} - Prophet Price Forecast for April 30, 2025')
        plt.xlabel('Date')
        plt.ylabel('Price')
        plt.legend()
        plt.grid(True)
        plt.xticks(rotation=45)
        plt.tight_layout()
        plt.savefig(f'{name}_prophet_forecast.png')
        plt.show()
        plt.close()

        # Interpretation of Prophet Forecast
        print(f'\nInterpretation of Prophet Forecast for {name}:')
        print(f'- Price Forecast: Models trends and daily seasonality in prices.')
        print(f'- Forecast Trend: {"Upward" if forecast_prices.iloc[-1] > forecast_prices.iloc[0] else "Downward or Flat"} trend expected.')
        print(f'- Reliability: Robust to trends and seasonality but may smooth out minute-by-minute volatility.')

    except Exception as e:
        print(f'Prophet forecast failed for {name}: {str(e)}')
